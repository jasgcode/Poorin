```python
# Copyright The Marin Authors
# SPDX-License-Identifier: Apache-2.0

"""Static import reachability for the Poorin prune.

Builds the first-party import graph of the workspace (every package under ``lib/*/src``, plus
``experiments`` and ``tools``), computes the closure of the roots in ``tools/roots.toml``,
explains why a module is reachable, and ranks modules by dominated weight: the lines that become
unreachable when that module alone is removed.

The default graph follows every import statement, including imports inside functions, so lazily
imported code is never reported as unreachable. ``--eager`` follows only imports that run when a
module is first imported (statements outside function bodies), which approximates startup cost.

Usage (from the repository root):
    uv run --no-sync python -m tools.reachability
    uv run --no-sync python -m tools.reachability --why levanter.eval_harness
    uv run --no-sync python -m tools.reachability --rank 25
    uv run --no-sync python -m tools.reachability --eager --rank 25
    uv run --no-sync python -m tools.reachability --json
"""

import argparse
import ast
import json
import tomllib
from collections import deque
from collections.abc import Collection, Iterable, Iterator, Mapping, Sequence
from dataclasses import asdict, dataclass
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[1]
ROOTS_FILE = REPO_ROOT / "tools" / "roots.toml"
TOP_LEVEL_PACKAGES = ("experiments", "tools")
TEST_DIR_NAMES = frozenset({"tests", "test"})
DEFERRED_SCOPES = (ast.FunctionDef, ast.AsyncFunctionDef, ast.Lambda)

Graph = dict[str, frozenset[str]]


@dataclass(frozen=True)
class RunnableRoot:
    """A root that ``prune_check`` executes: ``python -m <module> <args>``."""

    name: str
    module: str
    args: tuple[str, ...]


@dataclass(frozen=True)
class Roots:
    """The roots file: what must keep running, what must keep existing, what is deleted anyway."""

    runnable: tuple[RunnableRoot, ...]
    kept: tuple[str, ...]
    static_only: tuple[str, ...]
    deleted: tuple[str, ...]

    def patterns(self) -> tuple[str, ...]:
        """Every root pattern that contributes to the closure."""
        return (*(root.module for root in self.runnable), *self.kept, *self.static_only)


@dataclass(frozen=True)
class Summary:
    first_party_lines: int
    reachable_lines: int
    reachable_modules: int
    per_package: dict[str, tuple[int, int]]
    """Top-level package -> (reachable lines, total lines)."""


def load_roots(path: Path) -> Roots:
    data = tomllib.loads(path.read_text())
    return Roots(
        runnable=tuple(RunnableRoot(r["name"], r["module"], tuple(r["args"])) for r in data["runnable"]),
        kept=tuple(data["kept"]["modules"]),
        static_only=tuple(data["kept"]["static_only"]),
        deleted=tuple(data["deleted"]["modules"]),
    )


def is_test_file(rel_path: Path) -> bool:
    """Whether a repo-relative path is test code, which never counts as a module."""
    name = rel_path.name
    return (
        any(part in TEST_DIR_NAMES for part in rel_path.parts)
        or name.startswith("test_")
        or name.endswith("_test.py")
        or name == "conftest.py"
    )


def _source_roots(repo_root: Path) -> Iterator[tuple[Path, Path]]:
    """Yield (package directory, import root), both absolute."""
    for src in sorted(repo_root.glob("lib/*/src")):
        for package in sorted(src.iterdir()):
            if (package / "__init__.py").is_file():
                yield package, src
    for name in TOP_LEVEL_PACKAGES:
        if (repo_root / name / "__init__.py").is_file():
            yield repo_root / name, repo_root


def workspace_modules(repo_root: Path) -> dict[str, Path]:
    """Every non-test first-party module: dotted name -> file."""
    modules: dict[str, Path] = {}
    for package, import_root in _source_roots(repo_root):
        for py in sorted(package.rglob("*.py")):
            rel = py.relative_to(repo_root)
            if "__pycache__" in rel.parts or is_test_file(rel):
                continue
            parts = list(py.relative_to(import_root).with_suffix("").parts)
            if parts[-1] == "__init__":
                parts = parts[:-1]
            modules[".".join(parts)] = py
    return modules


def _ancestors(dotted: str) -> list[str]:
    parts = dotted.split(".")
    return [".".join(parts[: i + 1]) for i in range(len(parts))]


def _absolute_base(node: ast.ImportFrom, module: str, is_package: bool) -> str:
    """Absolute dotted prefix named by ``from ... import``. Copied from infra/ci/select_tests.py."""
    if node.level == 0:
        return node.module or ""
    package = module if is_package else module.rsplit(".", 1)[0] if "." in module else ""
    parts = package.split(".") if package else []
    up = node.level - 1
    if up > len(parts):
        return ""
    return ".".join(parts[: len(parts) - up] + (node.module.split(".") if node.module else []))


def _import_nodes(tree: ast.AST, *, eager: bool) -> Iterator[ast.Import | ast.ImportFrom]:
    stack: list[ast.AST] = [tree]
    while stack:
        node = stack.pop()
        if isinstance(node, ast.Import | ast.ImportFrom):
            yield node
        for child in ast.iter_child_nodes(node):
            if eager and isinstance(child, DEFERRED_SCOPES):
                continue
            stack.append(child)


def imported_modules(path: Path, module: str, known: Collection[str], *, eager: bool) -> set[str]:
    """Known modules that ``module``'s import statements execute, package ancestors included.

    ``from a.b import c`` contributes ``a.b`` and the candidate ``a.b.c``; only names that are
    real modules survive.
    """
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    is_package = path.name == "__init__.py"
    names: set[str] = set()
    for node in _import_nodes(tree, eager=eager):
        if isinstance(node, ast.Import):
            names.update(alias.name for alias in node.names)
            continue
        base = _absolute_base(node, module, is_package)
        if base:
            names.add(base)
            names.update(f"{base}.{alias.name}" for alias in node.names)
    return {ancestor for name in names for ancestor in _ancestors(name) if ancestor in known} - {module}


def build_graph(modules: Mapping[str, Path], *, eager: bool) -> Graph:
    known = set(modules)
    return {name: frozenset(imported_modules(path, name, known, eager=eager)) for name, path in modules.items()}


def expand(patterns: Iterable[str], modules: Collection[str]) -> set[str]:
    """Resolve root patterns to modules. A trailing ``.*`` selects every module under a package.

    A pattern that matches nothing raises: a stale roots file must not silently shrink the closure.
    """
    out: set[str] = set()
    for pattern in patterns:
        if pattern.endswith(".*"):
            prefix = pattern[:-2] + "."
            matched = {module for module in modules if module.startswith(prefix)}
        else:
            matched = {pattern} if pattern in modules else set()
        if not matched:
            raise ValueError(f"root pattern {pattern!r} matches no module")
        out |= matched
    return out


def _excluded_set(graph: Graph, excluded: Collection[str]) -> set[str]:
    return {m for m in graph if any(m == e or m.startswith(e + ".") for e in excluded)}


def closure(graph: Graph, roots: Iterable[str], excluded: Collection[str] = ()) -> set[str]:
    """Modules reachable from ``roots``. An excluded module, and every module under it, is not traversed."""
    blocked = _excluded_set(graph, excluded)
    seen: set[str] = set()
    stack = [root for root in roots if root not in blocked]
    while stack:
        module = stack.pop()
        if module in seen:
            continue
        seen.add(module)
        stack.extend(dep for dep in graph.get(module, ()) if dep not in seen and dep not in blocked)
    return seen


def import_chain(graph: Graph, roots: Iterable[str], target: str, excluded: Collection[str] = ()) -> list[str] | None:
    """Shortest import path from any root to ``target``, root first, or None if unreachable."""
    blocked = _excluded_set(graph, excluded)
    parent: dict[str, str | None] = {root: None for root in sorted(roots) if root not in blocked}
    queue = deque(parent)
    while queue:
        module = queue.popleft()
        if module == target:
            break
        for dep in sorted(graph.get(module, ())):
            if dep not in parent and dep not in blocked:
                parent[dep] = module
                queue.append(dep)
    if target not in parent:
        return None
    chain = [target]
    while (previous := parent[chain[-1]]) is not None:
        chain.append(previous)
    return chain[::-1]


def dominated_lines(
    graph: Graph, roots: Collection[str], lines: Mapping[str, int], excluded: Collection[str] = ()
) -> list[tuple[int, str]]:
    """For each reachable non-root module, the lines that become unreachable if it is removed.

    Recomputes the closure once per module. About 450 reachable modules take a few seconds, so a
    dominator-tree algorithm is not needed.
    """
    base = closure(graph, roots, excluded)
    weights = []
    for module in sorted(base - set(roots)):
        lost = base - closure(graph, roots, (*excluded, module))
        weights.append((sum(lines[m] for m in lost), module))
    return sorted(weights, reverse=True)


def line_counts(modules: Mapping[str, Path]) -> dict[str, int]:
    """Physical lines per module."""
    return {name: len(path.read_text(encoding="utf-8").splitlines()) for name, path in modules.items()}


def summarize(repo_root: Path, roots: Roots, *, eager: bool = False) -> Summary:
    modules = workspace_modules(repo_root)
    lines = line_counts(modules)
    graph = build_graph(modules, eager=eager)
    reached = closure(graph, expand(roots.patterns(), modules), roots.deleted)
    per_package: dict[str, tuple[int, int]] = {}
    for name, count in lines.items():
        package = name.split(".")[0]
        reachable, total = per_package.get(package, (0, 0))
        per_package[package] = (reachable + (count if name in reached else 0), total + count)
    return Summary(
        first_party_lines=sum(lines.values()),
        reachable_lines=sum(lines[m] for m in reached),
        reachable_modules=len(reached),
        per_package=per_package,
    )


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Static import reachability for the Poorin prune.")
    parser.add_argument("--eager", action="store_true", help="Follow only imports that run at import time.")
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument("--why", metavar="MODULE", help="Print the shortest import chain from a root to MODULE.")
    mode.add_argument("--rank", type=int, metavar="N", help="Print the N modules with the largest dominated weight.")
    mode.add_argument("--json", action="store_true", help="Print the summary as JSON.")
    args = parser.parse_args(argv)

    roots = load_roots(ROOTS_FILE)
    if args.json or (args.why is None and args.rank is None):
        summary = summarize(REPO_ROOT, roots, eager=args.eager)
        if args.json:
            print(json.dumps(asdict(summary), indent=2, sort_keys=True))
            return 0
        print(f"reachable: {summary.reachable_modules} modules, {summary.reachable_lines:,} of {summary.first_party_lines:,} lines")
        for package, (reachable, total) in sorted(summary.per_package.items(), key=lambda kv: -kv[1][1]):
            print(f"  {package:16s} {reachable:>9,} / {total:>9,}")
        return 0

    modules = workspace_modules(REPO_ROOT)
    graph = build_graph(modules, eager=args.eager)
    root_modules = expand(roots.patterns(), modules)
    if args.why is not None:
        chain = import_chain(graph, root_modules, args.why, roots.deleted)
        print("unreachable" if chain is None else "\n  <- ".join(reversed(chain)))
        return 0
    for weight, module in dominated_lines(graph, root_modules, line_counts(modules), roots.deleted)[: args.rank]:
        print(f"{weight:>8,}  {module}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
# Copyright The Marin Authors
# SPDX-License-Identifier: Apache-2.0

"""Run the Poorin prune invariant and append one row of measurements.

Run from the repository root before every prune commit:

    uv run --no-sync python -m tools.prune_check --phase P0 --step "short label"

Checks run in order and the first failure stops the run, so nothing is recorded:

1. Every kept root imports, except ``static_only`` roots.
2. ``pyrefly check`` passes.
3. ``./infra/pre-commit.py --changed-files`` passes.
4. The unit tests affected by the branch and working tree pass. Selection comes from
   ``infra/ci/select_tests.py``; pytest runs with ``--no-sync`` in the existing GPU venv and with
   ``JAX_PLATFORMS=cpu``, as in Marin's CI. ``infra/ci/run_tests.py`` is not used because it
   re-syncs ``.venv`` with CPU-only JAX.
5. Each runnable root trains to completion twice. The second run, with warm caches, is timed.

Runnable roots write under ``~/.cache/poorin/prune-check``. Per-user run outputs are deleted after
the runs; tokenized datasets and kernel autotune caches are kept so later checks stay warm.
"""

import argparse
import csv
import os
import shlex
import shutil
import subprocess
import sys
import sysconfig
import time
from collections.abc import Mapping, Sequence
from dataclasses import dataclass
from datetime import date
from pathlib import Path

from infra.ci.select_tests import select_local_tests
from tools.reachability import ROOTS_FILE, RunnableRoot, expand, load_roots, summarize, workspace_modules

REPO_ROOT = Path(__file__).resolve().parents[1]
METRICS_FILE = REPO_ROOT / "docs" / "poorin" / "prune-metrics.csv"
CACHE_HOME = Path.home() / ".cache" / "poorin"
MARIN_PREFIX = CACHE_HOME / "prune-check"
JAX_CACHE_DIR = CACHE_HOME / "jax"
BASE_REF = "main"

METRIC_COLUMNS = (
    "date",
    "phase",
    "step",
    "first_party_lines",
    "reachable_lines",
    "reachable_modules",
    "r1_imported_modules",
    "r1_import_time",
    "r1_warm_wall_time",
    "r2_warm_wall_time",
    "venv_size",
)


@dataclass(frozen=True)
class ImportProfile:
    modules: int
    seconds: float


def parse_importtime(stderr: str) -> ImportProfile:
    """Summarize ``python -X importtime`` output: modules imported and total time.

    Total time is the sum of cumulative times of top-level imports. Lines that are not import
    records, such as log lines that share stderr, are ignored.
    """
    modules = 0
    total_us = 0
    for line in stderr.splitlines():
        if not line.startswith("import time:"):
            continue
        fields = line.removeprefix("import time:").split("|")
        if len(fields) != 3 or not fields[0].strip().isdigit():
            continue  # the header row
        modules += 1
        name = fields[2]
        depth = (len(name) - len(name.lstrip(" ")) - 1) // 2
        if depth == 0:
            total_us += int(fields[1])
    return ImportProfile(modules=modules, seconds=total_us / 1e6)


def append_metrics_row(path: Path, row: Mapping[str, object]) -> None:
    """Append ``row`` to the metrics CSV, writing the header if the file is new."""
    if set(row) != set(METRIC_COLUMNS):
        raise ValueError(f"metrics row columns {sorted(row)} != {sorted(METRIC_COLUMNS)}")
    is_new = not path.exists()
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=METRIC_COLUMNS)
        if is_new:
            writer.writeheader()
        writer.writerow(row)


def remove_run_outputs(prefix: Path) -> None:
    """Delete per-user run outputs under ``prefix``. Shared datasets and caches stay."""
    users = prefix / "users"
    if users.exists():
        shutil.rmtree(users)


def directory_size(path: Path) -> int:
    return sum(f.stat().st_size for f in path.rglob("*") if f.is_file() and not f.is_symlink())


def run_env() -> dict[str, str]:
    """Environment for runnable roots. P5a moves the cache and NVRTC fixes into Levanter."""
    nvidia_lib = Path(sysconfig.get_paths()["purelib"]) / "nvidia" / "cu13" / "lib"
    env = dict(os.environ)
    env.update(
        MARIN_PREFIX=str(MARIN_PREFIX),
        JAX_COMPILATION_CACHE_DIR=str(JAX_CACHE_DIR),
        WANDB_MODE="offline",
        LD_LIBRARY_PATH=os.pathsep.join(p for p in (str(nvidia_lib), env.get("LD_LIBRARY_PATH", "")) if p),
    )
    return env


def _run(cmd: Sequence[str], *, env: Mapping[str, str] | None = None) -> None:
    print(f"$ {shlex.join(cmd)}", flush=True)
    subprocess.run(cmd, cwd=REPO_ROOT, env=env, check=True)


def _changed_files() -> list[str]:
    """Paths changed since the merge base with ``main``, including uncommitted and untracked files."""

    def git(*args: str) -> list[str]:
        out = subprocess.run(["git", *args], cwd=REPO_ROOT, check=True, capture_output=True, text=True).stdout
        return [line for line in out.splitlines() if line]

    merge_base = git("merge-base", BASE_REF, "HEAD")[0]
    return sorted(
        {*git("diff", "--name-only", merge_base), *git("ls-files", "--others", "--exclude-standard")}
    )


def run_affected_tests() -> None:
    selection = select_local_tests(_changed_files(), REPO_ROOT)
    paths = sorted({path for leg in selection.matrix for path in shlex.split(leg.test_paths)})
    if not paths:
        print("no affected tests", flush=True)
        return
    # CPU, as in CI: xdist starts many workers, and each would claim most of the GPU's memory.
    _run(["uv", "run", "--no-sync", "pytest", *paths], env={**os.environ, "JAX_PLATFORMS": "cpu"})


def import_kept_roots(kept: Sequence[str], static_only: Sequence[str], modules: Sequence[str]) -> None:
    targets = sorted(expand(kept, modules) - expand(static_only, modules))
    code = "import importlib, sys\nfor name in sys.argv[1:]:\n    importlib.import_module(name)\n"
    _run([sys.executable, "-c", code, *targets], env={**os.environ, "JAX_PLATFORMS": "cpu"})


def train(root: RunnableRoot, env: Mapping[str, str]) -> float:
    start = time.perf_counter()
    _run([sys.executable, "-m", root.module, *root.args], env=env)
    return time.perf_counter() - start


def profile_import(module: str) -> ImportProfile:
    result = subprocess.run(
        [sys.executable, "-X", "importtime", "-c", f"import {module}"],
        cwd=REPO_ROOT,
        env={**os.environ, "JAX_PLATFORMS": "cpu"},
        capture_output=True,
        text=True,
        check=True,
    )
    return parse_importtime(result.stderr)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Run the Poorin prune invariant.")
    parser.add_argument("--phase", required=True, help="Prune phase, for example P0.")
    parser.add_argument("--step", required=True, help="Short label for the change being checked.")
    parser.add_argument("--no-record", action="store_true", help="Run the checks without appending a row.")
    args = parser.parse_args(argv)

    roots = load_roots(ROOTS_FILE)
    modules = sorted(workspace_modules(REPO_ROOT))
    import_kept_roots(roots.kept, roots.static_only, modules)
    _run(["uv", "run", "--no-sync", "pyrefly", "check"])
    _run(["./infra/pre-commit.py", "--changed-files"])
    run_affected_tests()

    env = run_env()
    warm: dict[str, float] = {}
    for root in roots.runnable:
        train(root, env)  # first run fills the compile and autotune caches
        warm[root.name] = train(root, env)
    remove_run_outputs(MARIN_PREFIX)

    r1 = next(root for root in roots.runnable if root.name == "R1")
    profile = profile_import(r1.module)
    summary = summarize(REPO_ROOT, roots)
    row = {
        "date": date.today().isoformat(),
        "phase": args.phase,
        "step": args.step,
        "first_party_lines": summary.first_party_lines,
        "reachable_lines": summary.reachable_lines,
        "reachable_modules": summary.reachable_modules,
        "r1_imported_modules": profile.modules,
        "r1_import_time": round(profile.seconds, 2),
        "r1_warm_wall_time": round(warm["R1"], 1),
        "r2_warm_wall_time": round(warm["R2"], 1),
        "venv_size": directory_size(REPO_ROOT / ".venv"),
    }
    print(row, flush=True)
    if not args.no_record:
        append_metrics_row(METRICS_FILE, row)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def run_version(version: str, *, now: datetime, token: str) -> str:
    """The version a run uses for ``--version``.

    A mutable version (``dev`` or ``<label>-dev``) gets a run suffix so each invocation writes to
    a fresh output path: ``dev`` becomes ``20261005-091500-a1b2-dev`` and ``probe-dev`` becomes
    ``probe-20261005-091500-a1b2-dev``. Without the suffix, a rerun lands on the previous run's
    ``{name}/dev`` path and Levanter's checkpointer resumes from its last checkpoint instead of
    training from scratch. A calendar version is returned unchanged. ``--resume`` reuses a
    suffixed version to continue that run.
    """
    if not is_mutable_version(version):
        return version
    label = version.removesuffix("dev").rstrip("-")
    stamped = f"{now:%Y%m%d-%H%M%S}-{token}"
    return f"{label}-{stamped}-dev" if label else f"{stamped}-dev"
```

```python
_FA4_HEAD_DIMS = frozenset({64, 128})
_FA4_ARCH_FAMILIES = frozenset({8, 9, 10, 12})


def _fa4_supports(q: jax.Array, k: jax.Array, mask: object) -> bool:
    """Whether the default GPU path can use FA4: causal self-attention in the shapes its tests cover.

    FA4 handles causal masks, packed segment IDs, and sliding windows. It does not handle
    bidirectional masks, cross-attention, or float32 inputs. Head dimensions 64 and 128 and compute
    capabilities 8.x, 9.x, 10.x, and 12.x are the configurations covered by FA4's tests.
    """
    if not isinstance(mask, AttentionMask) or not mask.is_causal:
        return False
    if q.shape[1] != k.shape[1] or q.shape[-1] not in _FA4_HEAD_DIMS:
        return False
    if q.dtype not in (jnp.bfloat16, jnp.float16):
        return False
    from levanter.cutlass_kernel_cache import gpu_compute_capability  # noqa: PLC0415

    return gpu_compute_capability() // 10 in _FA4_ARCH_FAMILIES
```
